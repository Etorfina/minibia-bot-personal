(async () => {
  const bundleUrl = "https://raw.githubusercontent.com/Etorfina/minibia-bot-personal/develop/minibia-bot.js.gz.b64";
  const requiredControls = ['data-tab="healing"','id="minibia-bot-heal-spells"','id="minibia-bot-heal-items"','id="minibia-bot-heal-status"','id="minibia-bot-attack-target-list"','id="minibia-bot-attack-stance"','id="minibia-bot-attack-range"'];
  const response = await fetch(bundleUrl + "?v=" + Date.now(), { cache: "no-store" });
  if (!response.ok) throw new Error("No pude descargar el bot: HTTP " + response.status);
  const bytes = Uint8Array.from(atob((await response.text()).trim()), c => c.charCodeAt(0));
  const code = await new Response(new Blob([bytes]).stream().pipeThrough(new DecompressionStream("gzip"))).text();
  if (!requiredControls.every(control => code.includes(control))) throw new Error("La versión de develop no incluye los controles esperados.");
  window.minibiaBotSourceUrl = bundleUrl;
  (0, eval)(code);
  const fix = document.createElement("style");
  fix.id = "minibia-bot-collapsed-nav-fix";
  fix.textContent = '#minibia-bot-panel.mb-desktop .mb-mobile-tabs[hidden] { display: none !important; }';
  document.head.appendChild(fix);
  const placeToolbarAtTop = () => {
    const panel = document.getElementById("minibia-bot-panel");
    if (panel?.dataset.collapsed === "true") panel.style.top = "16px";
  };
  placeToolbarAtTop();
  document.getElementById("minibia-bot-collapse")?.addEventListener("click", () => requestAnimationFrame(placeToolbarAtTop));
  if (!document.querySelector('[data-tab="healing"]') || !document.getElementById("minibia-bot-heal-spells") || !document.getElementById("minibia-bot-heal-items") || !document.getElementById("minibia-bot-heal-status") || !document.getElementById("minibia-bot-attack-target-list") || !document.getElementById("minibia-bot-attack-stance") || !document.getElementById("minibia-bot-attack-range")) throw new Error("La versión se descargó, pero no apareció el panel. Recarga el juego y vuelve a ejecutar este código.");
  console.info("[minibia-bot] develop cargado con panel compacto arriba y pestañas ocultas al cerrar");
})().catch(error => { console.error(error); alert(error.message); });
